        title: 'Gallery App',
        desc: 'Photo gallery Android app with smooth loading, pinch-zoom, folder view and local caching using Room and Glide.',
        img: 'images/gallery-app.jpg',
        tags: ['Android','Kotlin','Room DB','Glide']
      },
      {
        id: 'note',
        title: 'Note App',
        desc: 'Lightweight note-taking app with tagging, search and offline persistence using Room Database and MVVM architecture.',
        img: 'images/note-app.jpg',
        tags: ['Android','Kotlin','Room DB','MVVM']
      },
    ];

    // DOM refs for slider
    const slideImage = document.getElementById('slideImage');
    const slideTitle = document.getElementById('slideTitle');
    const slideDesc = document.getElementById('slideDesc');
    const slideTags = document.getElementById('slideTags');
    const slideActions = document.getElementById('slideActions');
    const dotsContainer = document.getElementById('dots');
    const autoplayToggle = document.getElementById('autoplayToggle');
    const prevBtn = document.getElementById('prevBtn');
    const nextBtn = document.getElementById('nextBtn');

    let currentIndex = 0;
    let autoplay = false;
    let autoplayInterval = null;
    const AUTOPLAY_MS = 4000;

    // create dots
    projects.forEach((p, idx) => {
      const d = document.createElement('div');
      d.className = 'dot' + (idx === 0 ? ' active' : '');
      d.dataset.index = idx;
      d.addEventListener('click', () => goToSlide(idx));
      dotsContainer.appendChild(d);
    });

    function renderSlide(i){
      const p = projects[i];
      slideImage.src = p.img;
      slideImage.alt = p.title;
      slideTitle.textContent = p.title;
      slideDesc.textContent = p.desc;

      slideTags.innerHTML = '';
      p.tags.forEach(t => {
        const el = document.createElement('span');
        el.className = 'tag';
        el.textContent = t;
        slideTags.appendChild(el);
      });

      slideActions.innerHTML = '';
      const demo = document.createElement('a');
      demo.className = 'btn';
      demo.href = '#'; demo.target = "_blank";
      demo.textContent = 'View Demo';
      slideActions.appendChild(demo);

      const codeBtn = document.createElement('a');
      codeBtn.className = 'btn secondary';
      codeBtn.href = 'https://github.com/vickymvignesh56-arch'; codeBtn.target = "_blank";
      codeBtn.textContent = 'Source';
      slideActions.appendChild(codeBtn);

      document.querySelectorAll('.dot').forEach(d => d.classList.remove('active'));
      const activeDot = document.querySelector(`.dot[data-index="${i}"]`);
      if(activeDot) activeDot.classList.add('active');
    }

    function goToSlide(i){
      if(i < 0) i = projects.length - 1;
      if(i >= projects.length) i = 0;
      currentIndex = i;
      renderSlide(currentIndex);
      resetAutoplay();
    }

    prevBtn.addEventListener('click', () => goToSlide(currentIndex - 1));
    nextBtn.addEventListener('click', () => goToSlide(currentIndex + 1));

    autoplayToggle.addEventListener('click', () => {
      autoplay = !autoplay;
      autoplayToggle.textContent = autoplay ? 'Auto-play ON' : 'Auto-play OFF';
      if(autoplay) startAutoplay(); else stopAutoplay();
    });

    function startAutoplay(){
      stopAutoplay();
      autoplayInterval = setInterval(() => {
        goToSlide((currentIndex + 1) % projects.length);
      }, AUTOPLAY_MS);
    }
    function stopAutoplay(){
      if(autoplayInterval) {
        clearInterval(autoplayInterval);
        autoplayInterval = null;
      }
    }
    function resetAutoplay(){
      if(autoplay){
        stopAutoplay();
        startAutoplay();
      }
    }

    // initial render
    renderSlide(0);

    // graceful image fallback
    slideImage.addEventListener('error', () => {
      slideImage.style.objectFit = 'contain';
      slideImage.src = 'data:image/svg+xml;charset=UTF-8,' +
        encodeURIComponent(`<svg xmlns='http://www.w3.org/2000/svg' width='800' height='500'>
          <rect width='100%' height='100%' fill='#4a00e0'/>
          <text x='50%' y='50%' fill='#ffd7a6' font-family='Poppins, sans-serif' font-size='20' text-anchor='middle'>Replace image path: ${projects[currentIndex].img}</text>
        </svg>`);
    });

    // keyboard navigation
    window.addEventListener('keydown', (e) => {
      if(e.key === 'ArrowLeft') goToSlide(currentIndex - 1);
      if(e.key === 'ArrowRight') goToSlide(currentIndex + 1);
    });


  </script>
