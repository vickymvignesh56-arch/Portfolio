  <script>
    // Simple app state using localStorage
    const defaultState = {
      name: "Vignesh",
      title: "Web Developer",
      location: "Chennai, India",
      email: "you@example.com",
      phone: "+91 9xxxxxxxxx",
      links: "github.com/vignesh",
      about: "I build web apps using HTML, CSS and JavaScript.",
      skills: ["HTML","CSS","JavaScript"],
      photoURL: ""
    };

    function loadState(){
      try{
        const raw = localStorage.getItem('resume_state');
        return raw ? JSON.parse(raw) : defaultState;
      }catch(e){ return defaultState; }
    }

    function saveState(state){
      localStorage.setItem('resume_state', JSON.stringify(state));
      alert('Saved locally!');
    }

    // DOM refs
    const state = loadState();
    const editor = document.getElementById('editor');
    const editBtn = document.getElementById('editBtn');
    const saveBtn = document.getElementById('saveBtn');
    const pdfBtn = document.getElementById('pdfBtn');

    // input refs
    const nameI = document.getElementById('name');
    const titleI = document.getElementById('title');
    const locationI = document.getElementById('location');
    const emailI = document.getElementById('email');
    const phoneI = document.getElementById('phone');
    const linksI = document.getElementById('links');
    const aboutI = document.getElementById('about');
    const skillsI = document.getElementById('skills');
    const photoI = document.getElementById('photoURL');

    function render(){
      const s = loadState();
      document.getElementById('displayName').innerText = s.name;
      document.getElementById('displayTitle').innerText = `${s.title} • ${s.location}`;
      document.getElementById('contact').innerText = `${s.email} • ${s.phone} • ${s.links}`;
      document.getElementById('displayAbout').innerText = s.about;
      const skillsWrap = document.getElementById('skillsWrap');
      skillsWrap.innerHTML = '';
      (s.skills || []).forEach(sk=>{
        const d = document.createElement('span'); d.className='skill'; d.innerText = sk;
        skillsWrap.appendChild(d);
      });
      document.getElementById('photo').src = s.photoURL || 'data:image/svg+xml;utf8,<svg xmlns="http://www.w3.org/2000/svg" width="120" height="120"><rect width="100%" height="100%" fill="%23ddd"/><text x="50%" y="50%" dominant-baseline="middle" text-anchor="middle" fill="%23888" font-size="14">Avatar</text></svg>';
      // preview placeholders
      document.getElementById('experience').innerText = 'Experience and projects can be added later.';
      document.getElementById('education').innerText = 'Education info can be added later.';

      // fill editor inputs
      nameI.value = s.name; titleI.value = s.title; locationI.value = s.location; emailI.value = s.email;
      phoneI.value = s.phone; linksI.value = s.links; aboutI.value = s.about; skillsI.value = (s.skills||[]).join(', ');
      photoI.value = s.photoURL;
    }

    editBtn.addEventListener('click', ()=>{
      editor.style.display = editor.style.display === 'none' ? 'block' : 'none';
    });

    saveBtn.addEventListener('click', ()=>{
      const newState = {
        name: nameI.value||'',
        title: titleI.value||'',
        location: locationI.value||'',
        email: emailI.value||'',
        phone: phoneI.value||'',
        links: linksI.value||'',
        about: aboutI.value||'',
        skills: skillsI.value.split(',').map(s=>s.trim()).filter(Boolean),
        photoURL: photoI.value||''
      };
      saveState(newState);
      render();
    });

    pdfBtn.addEventListener('click', ()=>{
      // generate PDF of the card
      const element = document.querySelector('.card');
      const opt = {
        margin:       0.3,
        filename:     (state.name || 'resume') + '.pdf',
        image:        { type: 'jpeg', quality: 0.95 },
        html2canvas:  { scale: 2 },
        jsPDF:        { unit: 'in', format: 'a4', orientation: 'portrait' }
      };
      html2pdf().set(opt).from(element).save();
    });

    // initial render
    render();
  </script>
